P4 · Historisk FTE-supply per pool

cc_pool_supply_history — löser det statiska supply-problemet i analytiklagret.
✅ Exekverat (95 rader)
CREATE TABLE IF NOT EXISTS cc_pool_supply_history ( id BIGSERIAL PRIMARY KEY, pool_slug TEXT NOT NULL, year_month TEXT NOT NULL, fte_supply NUMERIC(6,2) NOT NULL, note TEXT, created_at TIMESTAMPTZ DEFAULT NOW(), UNIQUE(pool_slug,year_month)); INSERT INTO cc_pool_supply_history(pool_slug,year_month,fte_supply,note) SELECT slug,TO_CHAR(gs,'YYYY-MM'),fte_supply,'Seeded' FROM cc_pools CROSS JOIN generate_series(NOW()-INTERVAL'18 months',NOW(),INTERVAL'1 month') gs WHERE fte_supply IS NOT NULL ON CONFLICT DO NOTHING;

P3b · Historisk cc_ticket_meta-backfill

Stubbar för befintliga tickets i cc_ticket_events utan cc_ticket_meta-post.
✅ Exekverat (9 077 stubbar)
-- Utförd 2026-06-11: 9 077 cc_ticket_meta-stubbar skapades -- Från 8 933 tickets med pool_slug (minus 600 redan befintliga = 9 077 nettoinförda) -- Täckning: 6.7% → 101.6% (9 677/8 933 tickets berikade) -- Obs: reply_count/call_duration defaults till 0 — ingen Freshdesk-API-data -- cc-ticket-enricher uppdaterar dessa när nya tickets anländer via webhook INSERT INTO cc_ticket_meta (ticket_id, group_id, in_cc_scope, enriched_at, reply_count, agent_reply_count, is_answered) SELECT DISTINCT e.ticket_id, e.freshdesk_group_id::bigint, (e.pool_slug IS NOT NULL), NOW(), 0, 0, false FROM cc_ticket_events e WHERE e.pool_slug IS NOT NULL AND NOT EXISTS (SELECT 1 FROM cc_ticket_meta m WHERE m.ticket_id = e.ticket_id) ON CONFLICT (ticket_id) DO NOTHING;
